/** Separator alternation for a year range: symbols may be padded, words must be. */
function yearRangeSeparatorPattern(separators: string[]): string {
  const padding = '[ ._]';
  const words = separators.filter((s) => /^[a-z]+$/i.test(s));
  const symbols = separators.filter((s) => !/^[a-z]+$/i.test(s));
  const alternatives: string[] = [];
  if (symbols.length) alternatives.push(`${padding}*${buildOrPattern(symbols, undefined, true)}${padding}*`);
  if (words.length) alternatives.push(`${padding}+${buildOrPattern(words, undefined, true)}${padding}+`);
  return `(?:${alternatives.join('|')})`;
}
 
/** A range runs forward: "(2013-2005)" is two unrelated years, not a span. */
function isAscendingYearRange(match: Match): boolean {
  const children = match.children ? [...match.children] : [];
  const years = children.filter((c: Match) => c.name === 'year').map((c: Match) => Number(c.value));
  return years.length === 2 && years[0] < years[1];
}
 
/**
 * Group years into units: the two years of a range count as one, so they are kept or
 * removed together and a range is never released to the title.
 */
function yearUnits(years: Match[]): Match[][] {
  const units: Match[][] = [];
  const ranges = new Map<Match, Match[]>();
  for (const year of years) {
    const chain = year.tags.includes('year-range') ? rootParent(year) : undefined;
    if (!chain) {
      units.push([year]);
      continue;
    }
    const existing = ranges.get(chain);
    if (existing) {
      existing.push(year);
    } else {
      const unit = [year];
      ranges.set(chain, unit);
      units.push(unit);
    }
  }
  return units;
}
 
function rootParent(match: Match): Match | undefined {
  let root = match.parent;
  while (root?.parent) root = root.parent;
  return root;
}
 
/**
 * Rule to keep only the marked year in the filepart (remove others).
 * Prioritizes grouped years over ungrouped.
 */
class KeepMarkedYearInFilepart extends Rule {
  static override priority = 64;
  override consequence = RemoveMatch;
 
  override enabled(context: Context): boolean {
    return !isDisabled(context, 'year');
  }
 
  override when(matches: Matches, _context: Context): Match[] {
    const ret: Match[] = [];
    const yearMatches = matches.named('year') as Match[] | Match | undefined;
    const yearArray = Array.isArray(yearMatches) ? yearMatches : yearMatches ? [yearMatches] : [];
 
    if (yearArray.length <= 1) return ret;
 
    // Get path markers
    const pathMarkers = matches.markers.named('path') as Match[] | Match | undefined;
    const pathArray = Array.isArray(pathMarkers) ? pathMarkers : pathMarkers ? [pathMarkers] : [];
 
    for (const filepart of pathArray) {
      const yearsInPart = matches.range(
        filepart.start,
        filepart.end,
        (m) => m.name === 'year',
      ) as Match[] | Match | undefined;
      const yearsInPartArray = Array.isArray(yearsInPart) ? yearsInPart : yearsInPart ? [yearsInPart] : [];
 
      if (yearsInPartArray.length <= 1) continue;
 
      const units = yearUnits(yearsInPartArray);
      if (units.length <= 1) continue;
 
      const groupUnits: Match[][] = [];
      const ungroupUnits: Match[][] = [];
 
      for (const unit of units) {
        const grouped = unit.some(
          (year) => matches.markers.atMatch(year, (m: Match) => m.name === 'group', 0) as Match | undefined,
        );
        (grouped ? groupUnits : ungroupUnits).push(unit);
      }
 
      let removed: Match[][] = [];
      if (groupUnits.length > 0 && ungroupUnits.length > 0) {
        // Only remove ungrouped years that appear BEFORE the first grouped year
        // (they are likely part of the title). Ungrouped years after the grouped year
        // may be part of bonus_title or episode content.
        const firstGroupStart = groupUnits[0][0].start;
        removed = [
          ...ungroupUnits.filter((unit) => unit[0].start < firstGroupStart),
          ...groupUnits.slice(1),
        ];
      } else if (groupUnits.length === 0 && ungroupUnits[0].length > 1) {
        // A leading range is never released to the title: both its years are the year.
        removed = ungroupUnits.slice(1);
      } else if (groupUnits.length === 0) {
        removed = [ungroupUnits[0], ...ungroupUnits.slice(2)];
      }
      for (const unit of removed) ret.push(...unit);
    }
 
    return ret;
  }
}
 
/**
 * Remove years inside group markers when a strong SxxExx episode pattern exists.
 * e.g. "feud.s01e05.and.the.winner.is.(the.oscars.of.1963)" — 1963 is episode content, not year.
 */
class RemoveGroupedYearWithSxxExx extends Rule {
  static override priority = 64;
  override consequence = RemoveMatch;
 
  override when(matches: Matches, _context: Context): Match[] {
    const ret: Match[] = [];
    const hasSxxExx = (matches.named('episode') as Match[]).some(
      (m: Match) => m.tags?.includes('SxxExx')
    );
    if (!hasSxxExx) return ret;
 
    for (const year of (matches.named('year') as Match[])) {
      const group = matches.markers.atMatch(year, (m: Match) => m.name === 'group', 0) as Match | undefined;
      if (!group) continue;
      // Only remove if the group contains significant non-year content
      // e.g. "(the.oscars.of.1963)" has lots of text, but "(2005)" is just a year.
      // A year range fills its group with the other year and the separator, so it is
      // measured whole — otherwise "(2011-2017)" looks like text around a year.
      const range = year.tags.includes('year-range') ? rootParent(year) : undefined;
      const groupLen = group.end - group.start;
      const yearLen = range ? range.end - range.start : year.end - year.start;
      if (groupLen - yearLen > 4) {
        // Group has significant content beyond the year - it's episode title content
        ret.push(year);
      }
    }
    return ret;
  }
}
 
/**
 * Drop a date that is really a fragment of a longer dotted numeric run in the
 * title (upstream #963): "Evangelion.3.0.1.11" matches "0.1.11" as 2000-11-01
 * while the "3." in front of it belongs to the title. A digit claimed by another
 * property (the "1" of "S01" before a real "2015.10.13" stamp) does not count.
 *
 * The sibling shape — an unpadded "D.M.YYYY" like "Jackass.2.5.2007" — is
 * rejected earlier, inside searchDate, so the trailing "2007" still surfaces as
 * a year instead of being swallowed by the discarded date span.
 */
class RemoveDecimalVersionDate extends Rule {
  static override priority = 128;
  override consequence = RemoveMatch;
 
  override enabled(context: Context): boolean {
    return !isDisabled(context, 'date');
  }
 
  override when(matches: Matches, _context: Context): Match[] {
    const input: string = matches.inputString ?? '';
    const ret: Match[] = [];
 
    for (const dateMatch of (matches.named('date') as Match[])) {
      // Dot-joined only: a version run is written "3.0.1.11", while an
      // underscore stamp ("…Channel.5_2025_07_18…") is a genuine date whose
      // leading digit belongs to the token before it.
      if (!/^\d+(?:\.\d+)+$/.test(input.slice(dateMatch.start, dateMatch.end))) continue;
      const prevChar = input[dateMatch.start - 1];
      const prevDigit = input[dateMatch.start - 2];
      if (prevChar === '.' && prevDigit && /\d/.test(prevDigit)) {
        const claimed = matches.range(dateMatch.start - 2, dateMatch.start - 1,
          (m: Match) => !m.private) as Match[];
        if (!(Array.isArray(claimed) ? claimed.length : claimed ? 1 : 0)) {
          ret.push(dateMatch);
        }
      }
    }
 
    return ret;
  }
}
 
/**
 * Grow a broadcast/DVR date over its clock tail: in
 * "…Title_Channel.5_2025_07_18_21_00.mp4" (upstream #875, Mika3578) the
 * "_21_00" is the recording time, and left in a hole its digits are read as
 * episode numbers. Only a full "YYYY<sep>MM<sep>DD" stamp followed by two or
 * three two-digit groups on the same separator qualifies, and each group has to
 * be a valid hour/minute/second.
 */
class AbsorbBroadcastClockSuffix extends Rule {
  static override priority = 128;
  override consequence = RemoveMatch;
 
  override enabled(context: Context): boolean {
    return !isDisabled(context, 'date');
  }
 
  override when(matches: Matches, _context: Context): Array<[Match, number, Match[]]> | false {
    const input: string = matches.inputString ?? '';
    const ret: Array<[Match, number, Match[]]> = [];
 
    for (const dateMatch of (matches.named('date') as Match[])) {
      const stamp = /^\d{4}([-/ ._])\d{1,2}\1\d{1,2}$/.exec(
        input.slice(dateMatch.start, dateMatch.end));
      if (!stamp) continue;
 
      const sep = stamp[1].replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      const tail = new RegExp(`^((?:${sep}\\d{2}){2,3})(?!\\d)`)
        .exec(input.slice(dateMatch.end));
      if (!tail) continue;
 
      const units = tail[1].split(stamp[1]).filter(Boolean).map((u) => parseInt(u, 10));
      const limits = [23, 59, 59];
      if (units.some((u, i) => u > limits[i])) continue;
 
      const newEnd = dateMatch.end + tail[1].length;
      const claimed = (matches.range(dateMatch.end, newEnd,
        (m: Match) => !m.private) as Match[]) ?? [];
      ret.push([dateMatch, newEnd, Array.isArray(claimed) ? claimed : [claimed]]);
    }
 
    return ret.length ? ret : false;
  }
 
  override then(matches: any, whenResponse: Array<[Match, number, Match[]]>, _context: Context): void {
    for (const [dateMatch, newEnd, claimed] of whenResponse) {
      for (const stale of claimed) {
        const initiator = (stale as any).initiator ?? stale;
        for (const child of ((initiator as any).children?.toArray?.() ?? [])) matches.remove(child);
        matches.remove(initiator);
        matches.remove(stale);
      }
      matches.remove(dateMatch);
      dateMatch.end = newEnd;
      matches.append(dateMatch);
    }
  }
}
 
/**
 * Create a rebulk pattern for date detection.
 */
export function date(config: Record<string, unknown>): Rebulk {
  const rebulk = new Rebulk().defaults({ validator: sepsSurround });
 
  // Year regex
  rebulk.regex('\\d{4}', {
    name: 'year',
    formatter: (s: string) => parseInt(s, 10),
    disabled: (context: Context) => isDisabled(context, 'year'),
    conflictSolver: (match, other) =>
      other.name === 'episode' || other.name === 'season'
        ? other.raw && match.raw && other.raw.length < match.raw.length
          ? other
          : '__default__'
        : '__default__',
    validator: (m: Match) => sepsSurround(m) && validYear(m.value as number),
  });
 
  // A YYYY<sep>YYYY range is one year value, bracketed or not (upstream #988). Without
  // it each year was matched alone and arbitrated by KeepMarkedYearInFilepart, which
  // released the first one to the title — so `Movie 1994-2004` came back as
  // "Movie 1994" with year 2004, while the bracketed spelling happened to keep both.
  // The range is never expanded to the years between: it is a span of airing dates,
  // and a show on hiatus would collect years it never aired.
  const yearRangeSeparators = (config['year_range_separators'] as string[] | undefined) ?? ['-', '~', 'to', '–'];
  rebulk
    .chain({
      tags: ['year-range'],
      flags: 'i',
      children: true,
      privateParent: true,
      validateAll: true,
      validator: {
        __parent__: and_(sepsSurround, isAscendingYearRange),
        year: (m: Match) => validYear(Number(m.value)),
      },
      formatter: { year: (s: string) => parseInt(s, 10) },
      conflictSolver: (match, other) =>
        (other.name === 'episode' || other.name === 'season')
        && other.raw !== undefined && match.raw !== undefined
        && other.raw.length <= match.raw.length
          ? other
          : '__default__',
      disabled: (context: Context) => isDisabled(context, 'year'),
    })
    .defaults({ tags: ['year-range'], validator: null, privateParent: true })
    .regex('(?<year>\\d{4})')
    .regex(yearRangeSeparatorPattern(yearRangeSeparators) + '(?<year>\\d{4})');
 
  // Week regex
  const weekWords = config['week_words'] as string[] | undefined;
  if (weekWords && weekWords.length > 0) {
    const weekPattern = buildOrPattern(weekWords) + '-?(\\d{1,2})';
    rebulk.regex(weekPattern, {
      name: 'week',
      formatter: (s: string) => parseInt(s.replace(/\D/g, ''), 10),
      children: true,
      flags: 'i',
      abbreviations: [dash],
      conflictSolver: (match, other) =>
        other.name === 'episode' || other.name === 'season'
          ? other.raw && match.raw && other.raw.length < match.raw.length
            ? other
            : '__default__'
          : '__default__',
      validator: (m: Match) => sepsSurround(m) && validWeek(m.value as number),
    });
  }
 
  // Date functional pattern
  const dateFunctional = (inputString: string, context?: Context) => {
    const result = searchDate(
      inputString,
      context?.['date_year_first'] as boolean | undefined,
      context?.['date_day_first'] as boolean | undefined,
    );
    if (result) {
      return [[result[0], result[1], { value: result[2] }]];
    }
    return [];
  };
 
  rebulk.functional(dateFunctional, {
    name: 'date',
    properties: { date: [null] },
    disabled: (context: Context) => isDisabled(context, 'date'),
    conflictSolver: (match, other) =>
      other.name === 'episode' || other.name === 'season' || other.name === 'crc32'
        ? other
        : '__default__',
  });
 
  rebulk.rules(RemoveDecimalVersionDate, AbsorbBroadcastClockSuffix, KeepMarkedYearInFilepart, RemoveGroupedYearWithSxxExx, new AbsorbWeekdayPrefix((config['weekday_words'] as string[]) ?? []));
  return rebulk;
}
 
/**
 * Absorb a weekday word glued to a date into the date match (upstream #794).
 * "…S2025E01.Thu.2.Jan.2025.Mar.Menor.Spain…": "Thu" is the broadcast weekday, not
 * the title — growing the date span over it removes the bogus first title hole so
 * the real episode title after the date is reclaimed.
 */
class AbsorbWeekdayPrefix extends Rule {
  override consequence = RemoveMatch;
  private weekdayRe: RegExp;
 
  constructor(weekdayWords: string[]) {
    super();
    const sepsClass = '[' + seps.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + ']';
    this.weekdayRe = new RegExp(
      '(?:^|' + sepsClass + ')(' + buildOrPattern(weekdayWords.length ? weekdayWords : ['\\bnever\\b']) + ')' + sepsClass + '*$',
      'i',
    );
  }
 
  enabled(context: any): boolean {
    return !isDisabled(context, 'date');
  }
 
  when(matches: any, _context: any): any {
    const input: string = matches.inputString ?? '';
    const ret: Array<[any, number]> = [];
    for (const dateMatch of matches.named('date') ?? []) {
      const filepart = matches.markers.atMatch(dateMatch, (m: any) => m.name === 'path', 0);
      const lower = filepart ? filepart.start : 0;
      const segment = input.slice(lower, dateMatch.start);
      const weekday = this.weekdayRe.exec(segment);
      if (!weekday) continue;
      const groupStart = lower + weekday.index + weekday[0].indexOf(weekday[1]);
